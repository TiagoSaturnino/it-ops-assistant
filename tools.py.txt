def _winrm(hostname: str, script: str) -> str:
    import winrm  # pip install "pywinrm[kerberos]"
    sessao = winrm.Session(hostname, auth=(os.environ["WINRM_USER"], os.environ["WINRM_PASS"]), transport="kerberos")
    r = sessao.run_ps(script)
    return (r.std_out or r.std_err).decode("utf-8", "replace")

def diagnostico_windows(hostname: str, verificacao: str) -> Any:

    if MODO_SIMULACAO:
        return FIXTURES["windows"].get((hostname, verificacao), {"erro": "sem dados na simulação"})
    return _winrm(hostname, script)

def executar_ferramenta(nome: str, args: dict) -> str:
    fn = EXECUTORES.get(nome)
    if fn is None:
        return json.dumps({"erro": f"ferramenta desconhecida: {nome}"})
    try:
        return json.dumps(fn(**args), ensure_ascii=False, default=str)[:4000]  # limita o tamanho
    except Exception as exc:  # o erro volta ao modelo como dado
        return json.dumps({"erro": str(exc)})
